```python
for attempt in range(retry_count):
    try:
        response = await service_call()
        if response.status_code == 200:
            break
        elif attempt < retry_count - 1:
            print(f"{Colors.WARNING}↻{Colors.ENDC}", end="", flush=True)
            await asyncio.sleep(backoff)
        else:
            print(f"{Colors.WARNING}⚠{Colors.ENDC}")
    except Exception as e:
        if attempt < retry_count - 1:
            await asyncio.sleep(backoff)
        else:
            print(f"{Colors.FAIL}✗ {str(e)[:50]}{Colors.ENDC}")
```

#### 3. MCP Content Query
```python
async def query_mcp_for_content(self, topic: str, context: str, retry_count: int = 2) -> str:
    """Query MCP via gateway to generate content for a specific topic."""
    for attempt in range(retry_count):
        try:
            query_payload = {
                "mcp_id": self.mcp_id,
                "method": "POST",
                "path": "/api/query",
                "headers": {"Content-Type": "application/json"},
                "body": {
                    "query": f"Based on the training documents, provide detailed information about: {topic}. Context: {context}",
                    "use_local_llm": True,
                    "llm_model": "llama2"
                },
                "timeout_seconds": 30
            }
            
            response = await self.client.post(
                f"{self.services['mcp-gateway']}/api/v1/gateway/route",
                json=query_payload,
                headers={"X-Correlation-ID": self.correlation_id},
                timeout=30.0
            )
            
            if response.status_code == 200:
                data = response.json()
                # Extract content from response
                if isinstance(data.get("body"), dict):
                    return data["body"].get("answer", data["body"].get("response", str(data["body"])))
                return str(data.get("response", data.get("result", data)))
                
        except Exception as e:
            if attempt < retry_count - 1:
                await asyncio.sleep(1)
                continue
            return f"[Content generation unavailable: {str(e)[:100]}]"
    
    return "[Content generation failed after retries]"
```
